# GAME BASIC for SEGA SATURN (1998), main.bin loaded at 0x0600A000.
# Names and comments for gbcd_disasm.py. This file is our own analysis and holds
# no code or data from the CD, so it can live in the public repo.
#
# Format, one entry per line:
#   ADDR name                  label an address (code or data)
#   ADDR name ; text           label with a header comment
#   ADDR - ; text              a comment printed above the instruction at ADDR
#   .table ADDR kind count     a structured table (tokens, subkw, errptrs)
#   .section ADDR text         a banner printed before ADDR
# ADDR is hex without 0x. Anything after '#' at the start of a line is ignored.

.section 0600A000 Start-up, SCI serial driver and COM: link file commands
.section 0604A7E0 Backup RAM library (BUP)
.section 0604CFD0 C runtime library (libc)
.section 0605C7BF SEGA Graphics Library (SGL)
.section 06075A00 Interpreter variables and initialised data

.table 0609CBE0 tokens 288
.table 0609DEE0 subkw 104
.table 06075A80 errptrs 97

# --- start-up and system ---

0600E578 main_init ; Interpreter initialisation

0600A0E0 rx_drain_until_break ; Throw away received bytes until a break or timeout

0600A32C sci_send_byte ; Queue one byte for transmit (waits while tx_ring is full)

0600A520 keyq_push ; Push a key code into the key queue
0600A5E0 sci_send_buf ; Send a buffer of bytes
0600A60C timeout_check ; Has the link timeout (in VBLANK ticks) run out?
